import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: const HomePage(),
    );
  }
}

class HomePage extends StatelessWidget {
  const HomePage({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Menu'),
      ),
      body: Column(
        children: [
          const Expanded(
            child: Center(
              child: Text(
                'Conteúdo da página',
                style: TextStyle(fontSize: 22),
              ),
            ),
          ),

          Container(
            height: 80,
            color: Colors.blue,
            child: Row(
              // TESTE AS TRÊS OPÇÕES:
              // MainAxisAlignment.spaceBetween
              // → coloca o maior espaço possível ENTRE os itens,
              //   deixando pouco ou nenhum espaço nas extremidades.
              //
              // MainAxisAlignment.spaceAround
              // → coloca espaços ao redor de cada item.
              //   As extremidades ficam com metade do espaço.
              //
              // MainAxisAlignment.spaceEvenly
              // → distribui espaços IGUAIS entre os itens
              //   e também nas duas extremidades.
              //
              // Visualmente, spaceEvenly costuma deixar os três
              // itens com a distribuição mais uniforme.
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,

              children: [
                Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: const [
                    Icon(
                      Icons.home,
                      color: Colors.white,
                    ),
                    SizedBox(height: 4),
                    Text(
                      'Início',
                      style: TextStyle(color: Colors.white),
                    ),
                  ],
                ),

                Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: const [
                    Icon(
                      Icons.search,
                      color: Colors.white,
                    ),
                    SizedBox(height: 4),
                    Text(
                      'Pesquisar',
                      style: TextStyle(color: Colors.white),
                    ),
                  ],
                ),

                Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: const [
                    Icon(
                      Icons.person,
                      color: Colors.white,
                    ),
                    SizedBox(height: 4),
                    Text(
                      'Perfil',
                      style: TextStyle(color: Colors.white),
                    ),
                  ],
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}
