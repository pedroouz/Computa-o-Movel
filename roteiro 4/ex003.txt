import 'package:flutter/material.dart';

class Exercicio3 extends StatefulWidget {
  const Exercicio3({super.key});

  @override
  State<Exercicio3> createState() => _Exercicio3State();
}

class _Exercicio3State extends State<Exercicio3> {
  double opacidade = 0;

  void mostrarEsconder() {
    setState(() {
      opacidade = opacidade == 0 ? 1 : 0;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("Exercício 3"),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            AnimatedOpacity(
              duration: const Duration(seconds: 1),
              opacity: opacidade,
              child: const Text(
                "Olá, Flutter!",
                style: TextStyle(
                  fontSize: 30,
                ),
              ),
            ),

            const SizedBox(height: 20),

            ElevatedButton(
              onPressed: mostrarEsconder,
              child: const Text("Mostrar / Esconder"),
            ),
          ],
        ),
      ),
    );
  }
}
